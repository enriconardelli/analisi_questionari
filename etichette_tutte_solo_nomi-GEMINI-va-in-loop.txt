import numpy as np
import pandas as pd
from sklearn.cluster import AffinityPropagation
from sklearn.metrics.pairwise import cosine_similarity
from sentence_transformers import SentenceTransformer

#Load labels from the file
with open('etichette_tutte_solo_nomi.txt', 'r', encoding='utf-8') as f:
labels = [line.strip().lower() for line in f]

#Remove duplicates
labels = list(set(labels))

#Load the SentenceTransformer model
model = SentenceTransformer('paraphrase-MiniLM-L6-v2')

#Generate embeddings
embeddings = model.encode(labels)

#Calculate cosine similarity
similarity_matrix = cosine_similarity(embeddings)

#Create DataFrame for labels and their embeddings
df = pd.DataFrame({'label': labels, 'embedding': embeddings.tolist()})

print("Labels and embeddings ready for clustering.")&lt;ctrl98>
